from pathlib import Path 
import click
import config
import pandas as pd
from sklearn.model_selection import train_test_split

@click.command()
@click.argument("input_path",type=Path)
@click.argument("output_train_path",type=Path)
@click.argument("output_test_path",type=Path)
def split(
    input_path: Path,
    output_train_path: Path,
    output_test_path: Path,
    features: list = config.features,  # исправлено на list
    target: str = config.target,  # исправлено на str
):
    df = pd.read_csv(input_path)
    
    # Проверка наличия необходимых столбцов
    if not all(feature in df.columns for feature in features):
        raise ValueError("Отсутствуют некоторые признаки в датасете")
    if target not in df.columns:
        raise ValueError(f"Целевая колонка {target} не найдена")

    x_train, x_test, y_train, y_test = train_test_split(
        df[features], 
        df[target], 
        train_size=0.8,
        random_state=42
    )

    train = pd.concat([x_train, y_train], axis=1)
    test = pd.concat([x_test, y_test], axis=1)

    train.to_csv(output_train_path, index=False)
    test.to_csv(output_test_path, index=False)

if __name__ == "__main__":
    split.invoke()
